C++ Program: Fibonacci Using Iteration
#include <iostream>
using namespace std;

int fibonacciIterative(int n) {
    if (n == 0) return 0;
    if (n == 1) return 1;

    int a = 0, b = 1, c;
    for (int i = 2; i <= n; i++) {
        c = a + b;
        a = b;
        b = c;
    }
    return b;
}

C++ Program

Using Switch Case + Tracing
#include <iostream>
using namespace std;

void printArray(int arr[], int n) {
    for (int i = 0; i < n; i++)
        cout << arr[i] << " ";
    cout << endl;
}

int main() {
    int n, choice;
    cout << "Enter number of elements: ";
    cin >> n;

    int arr[n];
    cout << "Enter elements:\n";
    for (int i = 0; i < n; i++)
        cin >> arr[i];

    cout << "\nChoose Sorting Method:\n";
    cout << "1. Bubble Sort\n";
    cout << "2. Selection Sort\n";
    cout << "3. Insertion Sort\n";
    cout << "Enter choice: ";
    cin >> choice;

    switch (choice) {

    // ---------------- BUBBLE SORT ----------------
    case 1:
        cout << "\nTracing Bubble Sort:\n";
        for (int i = 0; i < n - 1; i++) {
            cout << "Pass " << i + 1 << ":\n";
            for (int j = 0; j < n - i - 1; j++) {
                cout << "Compare " << arr[j] << " and " << arr[j + 1];
                if (arr[j] > arr[j + 1]) {
                    cout << " -> Swap\n";
                    swap(arr[j], arr[j + 1]);
                } else {
                    cout << " -> No Swap\n";
                }
                printArray(arr, n);
            }
        }
        break;

    // ---------------- SELECTION SORT ----------------
    case 2:
        cout << "\nTracing Selection Sort:\n";
        for (int i = 0; i < n - 1; i++) {
            int minIndex = i;
            cout << "Pass " << i + 1 << ":\n";
            for (int j = i + 1; j < n; j++) {
                cout << "Compare " << arr[minIndex]
                     << " and " << arr[j] << endl;
                if (arr[j] < arr[minIndex]) {
                    minIndex = j;
                }
            }
            cout << "Swap " << arr[i]
                 << " and " << arr[minIndex] << endl;
            swap(arr[i], arr[minIndex]);
            printArray(arr, n);
        }
        break;

    // ---------------- INSERTION SORT ----------------
    case 3:
        cout << "\nTracing Insertion Sort:\n";
        for (int i = 1; i < n; i++) {
            int key = arr[i];
            int j = i - 1;
            cout << "Insert " << key << ":\n";
            while (j >= 0 && arr[j] > key) {
                cout << "Shift " << arr[j] << endl;
                arr[j + 1] = arr[j];
                j--;
                printArray(arr, n);
            }
            arr[j + 1] = key;
            printArray(arr, n);
        }
        break;

    default:
        cout << "Invalid choice!";
    }

    cout << "\nSorted Array:\n";
    printArray(arr, n);

    return 0;
}

Binary Search
Definition
Binary Search is a searching technique that works on a sorted array.
It repeatedly divides the search interval into two halves and compares the key with the middle element.

Theory
Array must be sorted

Compare key with middle element

If key < middle → search left half

If key > middle → search right half

Repeat until element is found or range becomes empty

#include <iostream>
using namespace std;

// ---------------- ITERATIVE BINARY SEARCH ----------------
int binarySearchIterative(int arr[], int n, int key) {
    int low = 0, high = n - 1, mid, step = 1;

    cout << "\nTracing Iterative Binary Search:\n";

    while (low <= high) {
        mid = (low + high) / 2;

        cout << "Step " << step++
             << ": low=" << low
             << " high=" << high
             << " mid=" << mid
             << " arr[mid]=" << arr[mid] << endl;

        if (arr[mid] == key) {
            cout << "Element found at index " << mid << endl;
            return mid;
        }
        else if (key < arr[mid]) {
            cout << "Key < arr[mid], search left half\n";
            high = mid - 1;
        }
        else {
            cout << "Key > arr[mid], search right half\n";
            low = mid + 1;
        }
    }

    cout << "Element not found\n";
    return -1;
}

// ---------------- RECURSIVE BINARY SEARCH ----------------
int binarySearchRecursive(int arr[], int low, int high, int key, int step = 1) {

    if (low > high) {
        cout << "Search space empty. Element not found\n";
        return -1;
    }

    int mid = (low + high) / 2;

    cout << "Step " << step
         << ": low=" << low
         << " high=" << high
         << " mid=" << mid
         << " arr[mid]=" << arr[mid] << endl;

    if (arr[mid] == key) {
        cout << "Element found at index " << mid << endl;
        return mid;
    }
    else if (key < arr[mid]) {
        cout << "Key < arr[mid], search left half\n";
        return binarySearchRecursive(arr, low, mid - 1, key, step + 1);
    }
    else {
        cout << "Key > arr[mid], search right half\n";
        return binarySearchRecursive(arr, mid + 1, high, key, step + 1);
    }
}
